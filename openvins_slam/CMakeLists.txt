cmake_minimum_required(VERSION 3.16)
project(openvins_slam)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosbag2_cpp REQUIRED)
find_package(rosbag2_storage REQUIRED)
find_package(mcap_vendor REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(ov_core REQUIRED)
find_package(ov_init REQUIRED)
find_package(ov_msckf REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs calib3d features2d)
find_package(yaml-cpp REQUIRED)
find_package(Boost REQUIRED COMPONENTS system filesystem thread date_time)
find_package(Ceres REQUIRED)

add_executable(slam_from_bag src/main.cpp src/calibration.cpp src/snapshot.cpp src/mcap_io.cpp src/trajectory_writer.cpp)
target_compile_definitions(slam_from_bag PRIVATE ROS_AVAILABLE=2 ENABLE_ARUCO_TAGS=1)
target_include_directories(slam_from_bag PRIVATE include ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(slam_from_bag ament_index_cpp rclcpp rosbag2_cpp rosbag2_storage mcap_vendor geometry_msgs nav_msgs sensor_msgs tf2_msgs rosgraph_msgs cv_bridge ov_core ov_init ov_msckf)
target_link_libraries(slam_from_bag mcap_vendor::mcap yaml-cpp Eigen3::Eigen ${OpenCV_LIBRARIES} ${Boost_LIBRARIES} Ceres::ceres)
install(TARGETS slam_from_bag DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md ACCEPTANCE.md DESTINATION share/${PROJECT_NAME})
install(PROGRAMS test/verify_mcap.py DESTINATION lib/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_trajectory_writer test/test_trajectory_writer.cpp src/trajectory_writer.cpp)
  target_include_directories(test_trajectory_writer PRIVATE include)
  ament_target_dependencies(test_trajectory_writer geometry_msgs)
  ament_add_gtest(test_calibration test/test_calibration.cpp src/calibration.cpp)
  target_compile_definitions(test_calibration PRIVATE ROS_AVAILABLE=2 ENABLE_ARUCO_TAGS=1)
  target_include_directories(test_calibration PRIVATE include ${OpenCV_INCLUDE_DIRS})
  ament_target_dependencies(test_calibration ament_index_cpp rclcpp ov_core ov_init ov_msckf)
  target_link_libraries(test_calibration yaml-cpp Eigen3::Eigen ${OpenCV_LIBRARIES} ${Boost_LIBRARIES} Ceres::ceres)
  ament_add_gtest(test_mcap_io test/test_mcap_io.cpp src/mcap_io.cpp)
  target_include_directories(test_mcap_io PRIVATE include)
  ament_target_dependencies(test_mcap_io rclcpp rosbag2_cpp rosbag2_storage mcap_vendor std_msgs tf2_msgs)
  target_link_libraries(test_mcap_io mcap_vendor::mcap)
  ament_add_gtest(test_snapshot test/test_snapshot.cpp src/snapshot.cpp)
  target_compile_definitions(test_snapshot PRIVATE ROS_AVAILABLE=2 ENABLE_ARUCO_TAGS=1)
  target_include_directories(test_snapshot PRIVATE include ${OpenCV_INCLUDE_DIRS})
  ament_target_dependencies(test_snapshot rclcpp geometry_msgs nav_msgs tf2_msgs ov_core ov_init ov_msckf)
  target_link_libraries(test_snapshot Eigen3::Eigen ${OpenCV_LIBRARIES} ${Boost_LIBRARIES} Ceres::ceres)
endif()
ament_package()
